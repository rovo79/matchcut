# Music Track and Beat Sync Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let the pipeline take a prerecorded music track, tell the narrative stage about it, snap cuts to its beats, and mix it under the source audio in the render.

**Architecture:** A new `audio.py` module analyses the track once with librosa (tempo, beats, downbeats, RMS energy) and caches the result as `music.json` in the workspace. The `Context` carries that analysis to the stages: the narrative prompt gains a music summary, a new `BeatSyncPass` (the last pass, after `polish`, so dissolves and polish trims cannot push cuts off the grid) moves cut points onto beats by adjusting unlocked clip ends, and `render.build_filter_graph` appends a ducking/mix/loudness chain when a track is configured. Everything is driven by a `MusicConfig` section and tested offline against a synthetic click track generated by ffmpeg.

**Tech Stack:** Python 3.9+, numpy, librosa 1.x (beat tracking only; decoding stays on ffmpeg), ffmpeg filters `atrim`, `afade`, `volume`, `asplit`, `sidechaincompress`, `amix`, `loudnorm`, pytest.

**Spec:** `docs/superpowers/specs/2026-10-15-audio-music.md` (Phase 1 section).

```python
def make_click_track(
    output: Path,
    seconds: float = 40.0,
    bpm: float = 120.0,
    beats_per_bar: int = 4,
    sample_rate: int = 22050,
) -> Path:
    """Synthetic music: a tick on every beat and a louder low thump on every bar (WAV)."""
    ffmpeg = _require("ffmpeg")
    beat = 60.0 / bpm
    bar = beat * beats_per_bar
    expr = (
        f"0.6*sin(2*PI*880*t)*lt(mod(t,{beat:.6f}),0.05)"
        f"+0.9*sin(2*PI*110*t)*lt(mod(t,{bar:.6f}),0.08)"
    )
    output.parent.mkdir(parents=True, exist_ok=True)
    run(
        [
            ffmpeg,
            "-v",
            "error",
            "-nostdin",
            "-y",
            "-f",
            "lavfi",
            "-t",
            f"{seconds:.3f}",
            "-i",
            f"aevalsrc='{expr}':s={sample_rate}",
            "-c:a",
            "pcm_s16le",
            str(output),
        ],
        capture=False,
    )
    return output
```

```python
"""Music track analysis: tempo, beats, downbeats and an energy curve.

Decoding goes through ffmpeg (same as the rest of the pipeline); librosa is
used only for onset strength and beat tracking.
"""
from __future__ import annotations

import json
import subprocess
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Dict, List, Optional

import numpy as np

from .config import MusicConfig
from .media import MediaError, _require

ONSET_HOP = 512  # librosa hop length in samples


def decode_audio(path: str, sample_rate: int, start: float = 0.0, duration: Optional[float] = None) -> np.ndarray:
    """Mono float32 samples in -1..1 at ``sample_rate``."""
    ffmpeg = _require("ffmpeg")
    cmd: List[str] = [ffmpeg, "-v", "error", "-nostdin"]
    if start > 0:
        cmd += ["-ss", f"{start:.3f}"]
    cmd += ["-i", str(path)]
    if duration is not None:
        cmd += ["-t", f"{duration:.3f}"]
    cmd += ["-vn", "-ac", "1", "-ar", str(sample_rate), "-f", "f32le", "-"]
    proc = subprocess.run(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE, check=False)
    if proc.returncode != 0 or not proc.stdout:
        err = proc.stderr.decode("utf-8", "replace")[-500:]
        raise MediaError(f"could not decode audio from {path}: {err}")
    return np.frombuffer(proc.stdout, dtype=np.float32).copy()


@dataclass
class MusicAnalysis:
    path: str
    duration: float
    sample_rate: int
    tempo_bpm: float
    beats_per_bar: int
    beats: np.ndarray  # seconds into the track, ascending
    downbeats: np.ndarray  # subset of beats (every beats_per_bar-th, phase by onset strength)
    energy: np.ndarray  # RMS per hop, normalised so the loudest hop is 1.0
    hop_seconds: float

    # -- queries ---------------------------------------------------------
    def nearest_beat(self, t: float, downbeats: bool = False) -> Optional[float]:
        grid = self.downbeats if downbeats else self.beats
        if len(grid) == 0:
            return None
        return float(grid[int(np.argmin(np.abs(grid - t)))])

    def energy_at(self, t: float) -> float:
        if len(self.energy) == 0:
            return 0.0
        i = int(np.clip(int(t / self.hop_seconds), 0, len(self.energy) - 1))
        return float(self.energy[i])

    def summary(self, every: float = 5.0) -> Dict[str, Any]:
        """Compact description for prompts: tempo, length, energy curve, rises and drops."""
        n = int(self.duration // every) + 1
        curve = [round(self.energy_at(k * every), 2) for k in range(n)]
        rises: List[float] = []
        drops: List[float] = []
        for k in range(1, n):
            delta = curve[k] - curve[k - 1]
            if delta >= 0.25:
                rises.append(k * every)
            elif delta <= -0.25:
                drops.append(k * every)
        return dict(
            tempo_bpm=round(self.tempo_bpm, 1),
            duration_s=round(self.duration, 1),
            beats_per_bar=self.beats_per_bar,
            energy_every_s=every,
            energy_curve=curve,
            rises_at_s=rises,
            drops_at_s=drops,
        )

    # -- serialisation ---------------------------------------------------
    def to_dict(self) -> dict:
        return dict(
            path=self.path,
            duration=self.duration,
            sample_rate=self.sample_rate,
            tempo_bpm=self.tempo_bpm,
            beats_per_bar=self.beats_per_bar,
            beats=[round(float(b), 4) for b in self.beats],
            downbeats=[round(float(b), 4) for b in self.downbeats],
            energy=[round(float(e), 4) for e in self.energy],
            hop_seconds=self.hop_seconds,
        )

    @classmethod
    def from_dict(cls, d: dict) -> "MusicAnalysis":
        return cls(
            path=str(d["path"]),
            duration=float(d["duration"]),
            sample_rate=int(d["sample_rate"]),
            tempo_bpm=float(d["tempo_bpm"]),
            beats_per_bar=int(d.get("beats_per_bar", 4)),
            beats=np.array(d["beats"], dtype=np.float32),
            downbeats=np.array(d["downbeats"], dtype=np.float32),
            energy=np.array(d["energy"], dtype=np.float32),
            hop_seconds=float(d["hop_seconds"]),
        )

    def save(self, path: Path) -> None:
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(json.dumps(self.to_dict(), indent=2))

    @classmethod
    def load(cls, path: Path) -> "MusicAnalysis":
        return cls.from_dict(json.loads(Path(path).read_text()))


def analyse_music(path: str, cfg: MusicConfig) -> MusicAnalysis:
    try:
        import librosa
    except ImportError as e:  # pragma: no cover
        raise MediaError("pip install librosa to analyse a music track") from e

    sr = cfg.sample_rate
    y = decode_audio(path, sr)
    duration = len(y) / sr

    onset = librosa.onset.onset_strength(y=y, sr=sr, hop_length=ONSET_HOP)
    tempo, beat_frames = librosa.beat.beat_track(onset_envelope=onset, sr=sr, hop_length=ONSET_HOP, units="frames")
    tempo_bpm = float(np.atleast_1d(tempo)[0]) if np.size(tempo) else 0.0
    beat_frames = np.asarray(beat_frames, dtype=int)
    beats = librosa.frames_to_time(beat_frames, sr=sr, hop_length=ONSET_HOP).astype(np.float32)

    bpb = max(1, int(cfg.beats_per_bar))
    if len(beat_frames):
        strength = onset[np.clip(beat_frames, 0, len(onset) - 1)]
        phase = int(np.argmax([float(strength[k::bpb].sum()) for k in range(bpb)]))
        downbeats = beats[phase::bpb]
    else:
        downbeats = beats

    hop = max(1, int(sr * cfg.hop_seconds))
    n = len(y) // hop
    if n:
        frames = y[: n * hop].reshape(n, hop)
        rms = np.sqrt(np.mean(frames * frames, axis=1))
        energy = (rms / max(float(rms.max()), 1e-6)).astype(np.float32)
    else:
        energy = np.zeros(0, dtype=np.float32)

    return MusicAnalysis(
        path=str(path),
        duration=duration,
        sample_rate=sr,
        tempo_bpm=tempo_bpm,
        beats_per_bar=bpb,
        beats=beats,
        downbeats=np.asarray(downbeats, dtype=np.float32),
        energy=energy,
        hop_seconds=cfg.hop_seconds,
    )
```

```python
# ==========================================================================
# beat sync
# ==========================================================================
class BeatSyncPass(Pass):
    """Move every cut onto the nearest beat of the music track.

    The cut between clips i-1 and i happens at ``record_start(i)`` on the
    timeline (for a dissolve: where the incoming clip starts fading in), i.e.
    ``record_start(i) + music.start`` seconds into the track. Only the outgoing
    clip can move it: its out-point when unlocked, else its in-point (a shorter
    head lands the junction earlier). Locked ends stay. The pass runs last so
    nothing moves a cut after it is on the beat.
    """

    name = "beatsync"

    def apply(self, timeline: Timeline, ctx: Context) -> PassResult:
        tl = timeline.copy()
        decisions: List[Decision] = []
        music = ctx.music
        mcfg = ctx.cfg.music
        if music is None or not mcfg.snap:
            tl.stage = self.name
            return PassResult(tl, decisions)

        for i in range(1, len(tl.clips)):
            prev, cur = tl.clips[i - 1], tl.clips[i]
            track_t = tl.record_start(i) + mcfg.start
            beat_change = prev.beat_id != cur.beat_id
            target: Optional[float] = None
            downbeat = False
            if beat_change and mcfg.prefer_downbeats_at_beat_changes:
                db = music.nearest_beat(track_t, downbeats=True)
                if db is not None and abs(db - track_t) <= mcfg.snap_window:
                    target, downbeat = db, True
            if target is None:
                b = music.nearest_beat(track_t)
                if b is not None and abs(b - track_t) <= mcfg.snap_window:
                    target = b
            if target is None:
                decisions.append(self.decision("skip", [prev.id, cur.id], f"no beat within {mcfg.snap_window:.2f}s of the cut"))
                continue
            delta = target - track_t
            if abs(delta) < 0.02:
                decisions.append(self.decision("on_beat", [prev.id, cur.id], "cut already on the beat", 0.0))
                continue
            if self._shift_junction(prev, delta, ctx):
                decisions.append(
                    self.decision(
                        "snap",
                        [prev.id, cur.id],
                        f"cut moved {delta:+.3f}s onto a {'downbeat' if downbeat else 'beat'} at {target:.2f}s of the track",
                        abs(delta),
                        delta=round(delta, 4),
                        beat=round(float(target), 4),
                        downbeat=downbeat,
                    )
                )
            else:
                decisions.append(self.decision("locked", [prev.id, cur.id], "both ends of the outgoing clip are locked or out of room"))
        tl.stage = self.name
        tl.history.extend(decisions)
        return PassResult(tl, decisions)

    @staticmethod
    def _shift_junction(clip: Clip, delta: float, ctx: Context) -> bool:
        """Move the junction after ``clip`` by ``delta`` seconds via one of its unlocked ends."""
        s, e = ctx.shot_bounds[clip.shot_index]
        pacing = ctx.cfg.pacing
        if not clip.locked_out:
            new_out = clip.source_out + delta
            if s <= new_out <= e and pacing.min_clip_seconds <= new_out - clip.source_in <= pacing.max_clip_seconds:
                clip.source_out = round(new_out, 3)
                return True
        if not clip.locked_in:
            new_in = clip.source_in - delta
            if s <= new_in <= e and pacing.min_clip_seconds <= clip.source_out - new_in <= pacing.max_clip_seconds:
                clip.source_in = round(new_in, 3)
                return True
        return False
```

```python
def build_filter_graph(tl: Timeline, media: MediaInfo, cfg: RenderConfig, music: Optional[MusicConfig] = None) -> str:
    """ffmpeg -filter_complex graph assembling the timeline from input 0 (and music from input 1)."""
    fps = media.fps
    width = cfg.proxy_width if cfg.proxy else cfg.width
    scale = f",scale={width}:-2:flags=bicubic" if width else ""
    parts: List[str] = []
    has_music = bool(music is not None and music.path)
    audio = media.has_audio and not (has_music and music.only)
    for i, c in enumerate(tl.clips):
        parts.append(
            f"[0:v]trim=start={c.source_in:.4f}:end={c.source_out:.4f},setpts=PTS-STARTPTS,"
            f"fps={fps:.4f},settb=AVTB,format=yuv420p,setsar=1{scale}[v{i}]"
        )
        if audio:
            parts.append(f"[0:a]atrim=start={c.source_in:.4f}:end={c.source_out:.4f},asetpts=PTS-STARTPTS,asettb=AVTB[a{i}]")

    cur_v, cur_a = "[v0]", "[a0]"
    cur_dur = tl.clips[0].duration
    for i, c in enumerate(tl.clips[1:], start=1):
        tr = c.transition_in
        nv, na = f"[xv{i}]", f"[xa{i}]"
        if tr.is_cut:
            parts.append(f"{cur_v}[v{i}]concat=n=2:v=1:a=0{nv}")
            if audio:
                parts.append(f"{cur_a}[a{i}]concat=n=2:v=0:a=1{na}")
            cur_dur += c.duration
        else:
            kind = "fadeblack" if tr.kind == "dip_to_black" else "fade"
            offset = max(0.0, cur_dur - tr.duration)
            parts.append(f"{cur_v}[v{i}]xfade=transition={kind}:duration={tr.duration:.3f}:offset={offset:.3f}{nv}")
            if audio:
                parts.append(f"{cur_a}[a{i}]acrossfade=d={tr.duration:.3f}{na}")
            cur_dur += c.duration - tr.duration
        cur_v, cur_a = nv, na

    vfilters = []
    afilters = []
    if tl.fade_in > 0:
        vfilters.append(f"fade=t=in:st=0:d={tl.fade_in:.3f}")
        afilters.append(f"afade=t=in:st=0:d={tl.fade_in:.3f}")
    if tl.fade_out > 0:
        st = max(0.0, cur_dur - tl.fade_out)
        vfilters.append(f"fade=t=out:st={st:.3f}:d={tl.fade_out:.3f}")
        afilters.append(f"afade=t=out:st={st:.3f}:d={tl.fade_out:.3f}")
    parts.append(f"{cur_v}{','.join(vfilters) if vfilters else 'null'}[vout]")
    if audio:
        label = "[asrc]" if has_music else "[aout]"
        parts.append(f"{cur_a}{','.join(afilters) if afilters else 'anull'}{label}")
    if has_music:
        parts += _music_chain(music, total=cur_dur, source_audio=audio)
    return ";".join(parts)


def _music_chain(music: MusicConfig, total: float, source_audio: bool) -> List[str]:
    """Trim, fade, gain, duck (optional), mix and normalise the music (input 1) under [asrc]."""
    parts: List[str] = []
    fade = min(music.fade_out, total / 2)
    start = max(0.0, music.start)
    parts.append(
        f"[1:a]atrim=start={start:.3f}:end={start + total:.3f},asetpts=PTS-STARTPTS,"
        f"afade=t=out:st={max(0.0, total - fade):.3f}:d={fade:.3f},volume={music.gain_db:.1f}dB[music]"
    )
    if source_audio:
        if music.duck:
            parts.append("[asrc]asplit[asrc0][asrc1]")
            parts.append(
                f"[music][asrc1]sidechaincompress=threshold={music.duck_threshold}:ratio={music.duck_ratio}:attack=20:release=300[ducked]"
            )
            parts.append("[asrc0][ducked]amix=inputs=2:duration=first:normalize=0[amixed]")
        else:
            parts.append("[asrc][music]amix=inputs=2:duration=first:normalize=0[amixed]")
    else:
        parts.append("[music]anull[amixed]")
    final = f"loudnorm=I={music.target_lufs:.1f}:TP=-1.5:LRA=11" if music.loudnorm else "anull"
    parts.append(f"[amixed]{final}[aout]")
    return parts


def render(
    tl: Timeline,
    media: MediaInfo,
    cfg: RenderConfig,
    output: Path,
    overwrite: bool = True,
    music: Optional[MusicConfig] = None,
) -> Path:
    if not tl.clips:
        raise ValueError("cannot render an empty timeline")
    ffmpeg = _require("ffmpeg")
    has_music = bool(music is not None and music.path)
    graph = build_filter_graph(tl, media, cfg, music=music)
    output.parent.mkdir(parents=True, exist_ok=True)
    cmd = [ffmpeg, "-v", "error", "-nostdin", "-y" if overwrite else "-n", "-i", media.path]
    if has_music:
        cmd += ["-i", str(music.path)]
    cmd += ["-filter_complex", graph, "-map", "[vout]"]
    if media.has_audio or has_music:
        cmd += ["-map", "[aout]", "-c:a", "aac", "-b:a", cfg.audio_bitrate]
    cmd += [
        "-c:v",
        "libx264",
        "-crf",
        str(cfg.crf if not cfg.proxy else max(cfg.crf, 26)),
        "-preset",
        "veryfast" if cfg.proxy else cfg.preset,
        "-pix_fmt",
        "yuv420p",
        "-movflags",
        "+faststart",
        str(output),
    ]
    log.info("rendering %d clips (%.1fs)%s -> %s", len(tl.clips), tl.duration, " with music" if has_music else "", output)
    run(cmd, capture=False)
    return output
```
